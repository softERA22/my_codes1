import java.util.*;
public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    // n =n*-1;
     if(n<0){
      n = n*-1;
    }
    int rev  = 0;
    while(n>0){
     
      int d = n%10;
      rev =  rev*10+d;
      n = n/10;
   
      
    }
    
    System.out.print(rev );
    
  }
}
