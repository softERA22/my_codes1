import java.util.*;
public class Main {
  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
    int n = sc.nextInt();
    if (n<0){
      n=n*-1;
    }
    int sum = 0;
    while(n>0){
      
       int d = n%10;
       sum = sum+d;
       n=n/10;
    }
      System.out.println(sum);
      
      
    
      System.out.println("Hello, World!");
  }
}
