import java.util.*;
public class Main {

  static void printarrr(int[] ar ){
    int n = ar.length;
     int lv = ar[n-1];
    
    for(int i = n-1; i>=1; i--){
      ar[i] = ar[i-1];
      }
      
       ar[0] = lv;
  

      for(int i =0; i<n; i++ ){
        System.out.print(ar[i] + " ");
      }
   
     
  
  }


  public static void main(String[] args) {
    Scanner sc = new Scanner(System.in);
     int n = sc.nextInt();
       int ar[] = new int[n];
      
      for(int i = 0; i<n; i++){
        ar[i] = sc.nextInt();
      }
      

      printarrr(ar);
       

      
  }
}
