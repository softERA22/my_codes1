import java.util.*;
public class Main {

  static void printarr(int[] ar ){

    int n = ar.length;
    
     for(int i = 0; i<n; i++){
        
       int sum = 0;
      while(sum<=i){
        sum++;
      }
          System.out.println(sum + " " +(ar[i]));
          //  System.out.println(ar[i] );
          }
      }



  public static void main(String[] args) {
    
      Scanner sc = new Scanner(System.in);
      int q = sc.nextInt();
      int ar[] = new int[q];
      
      for(int i = 0; i<q; i++){
        ar[i] = sc.nextInt();
          
      }
      
      printarr(ar);
  }
}
