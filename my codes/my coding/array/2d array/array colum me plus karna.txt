import java.util.*;
public class Main {
    static void  printMatrix(int[][] matrix) {
      int n = matrix.length;
      int m = matrix[0].length;
      int[] ans = new int[m];
      
            for(int j = 0; j<m; j++){
              int sum = 0;
              
               for(int i =0; i<n; i++){
                sum = sum+matrix[i][j];
                
               }
               ans[j] = sum; 
               System.out.print(ans[j] + " ");
            }
           
    }
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int m = sc.nextInt();

        int[][] matrix = new int[n][m];
        
        for(int i = 0; i < n; i++) {
            for(int j = 0; j < m; j++) {
                matrix[i][j] = sc.nextInt();
            }
        }
         printMatrix(matrix);     
       
    }
}
