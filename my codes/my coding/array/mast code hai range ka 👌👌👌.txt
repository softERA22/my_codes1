public class Main {
  public static void main(String[] args) {
      int[] ar = {1,2,3,4,5,6,};
        
        int i = 1;
        int j = 5;

        while(i<j){
          int temp = ar[i];
          ar[i] = ar[j];
          ar[j] = temp;
          i++;
          j--;
        }
        System.out.println(ar[0]);
        System.out.println(ar[1]);
        System.out.println(ar[2]);
        System.out.println(ar[3]);
        System.out.println(ar[4]);
        System.out.println(ar[5]);

  }
}
